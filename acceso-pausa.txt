<script>
  // Hash SHA-256 de la contraseña (la contraseña no aparece escrita aquí)
  const HASH = "89ef11c9a1ecee9cdadb32501a30f65b2551543321b21c1877c218599f0ba3cf";
  const MENU = "index.html";

  async function sha256(text) {
    const data = new TextEncoder().encode(text);
    const buf = await crypto.subtle.digest("SHA-256", data);
    return Array.from(new Uint8Array(buf))
      .map(b => b.toString(16).padStart(2, "0")).join("");
  }

  const input = document.getElementById("pass");
  const msg = document.getElementById("msg");

  input.addEventListener("keydown", async (e) => {
    if (e.key !== "Enter") return;
    if (await sha256(input.value) === HASH) {
      sessionStorage.setItem("acceso", "ok");
      location.href = MENU;
    } else {
      msg.textContent = "Contraseña incorrecta";
      input.value = "";
    }
  });
</script>
